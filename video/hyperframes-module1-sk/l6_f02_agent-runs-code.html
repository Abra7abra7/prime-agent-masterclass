import pandas as pd
import matplotlib.pyplot as plt

# Načítanie marketingových dát
df = pd.read_csv('data/marketing.csv')

# Analýza konverzií podľa kanála
conv = df.groupby('channel')['conversions'].sum()
# channel
# email       1,247
# organic     3,891
# paid        2,103
# social        856

# Vizualizácia
conv.plot(kind='bar', title='Konverzie podľa kanála')
plt.savefig('output/chart.png', dpi=150)
