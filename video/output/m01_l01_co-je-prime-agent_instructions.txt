## ČASŤ C: FFMPEG SPOJENIE

# ── FFmpeg: Finálne spojenie ────────────────────────────────────────────
# Variant A: Avatar vľavo (malý), screen demo vpravo
ffmpeg -i "/Users/abra/Developer/prime-agent-masterclass/video/output/m01_l01_co-je-prime-agent_avatar.mp4" -i "/Users/abra/Developer/prime-agent-masterclass/video/output/m01_l01_co-je-prime-agent_screen_full.mp4" \
  -filter_complex "
    [0:v]scale=480:270,format=yuv420p[avatar];
    [1:v]scale=1440:810,format=yuv420p[screen];
    [avatar]pad=1920:1080:20:810:black[left];
    [left][screen]overlay=480:0[out]
  " \
  -map "[out]" -map 1:a \
  -c:v libx264 -preset medium -crf 23 \
  -c:a aac -b:a 128k \
  -shortest \
  "/Users/abra/Developer/prime-agent-masterclass/video/output/m01_l01_co-je-prime-agent_final.mp4"

# Variant B: Avatar + screen vedľa seba (50/50 split)
